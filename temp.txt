param(
  [Parameter(Mandatory=$true)]
  [string]$LoginName = 'konto\domena',

  [string]$ServersFile = ".\servers.txt",
  [string]$OutDir = ".\out",
  [switch]$UseIntegratedSecurity = $true,

  # jeśli używasz SQL auth, podaj -SqlUser i -SqlPassword
  [string]$SqlUser,
  [string]$SqlPassword
)

# --- wymagane: moduł SqlServer (Invoke-Sqlcmd) ---
if (-not (Get-Module -ListAvailable -Name SqlServer)) {
  throw "Brak modułu 'SqlServer'. Zainstaluj: Install-Module SqlServer -Scope CurrentUser"
}

New-Item -ItemType Directory -Force -Path $OutDir | Out-Null

$servers = Get-Content $ServersFile | Where-Object { $_.Trim() -and -not $_.Trim().StartsWith("#") } | ForEach-Object { $_.Trim() }

$tsql = @"
SET NOCOUNT ON;

DECLARE @Login sysname = N'$(LoginName)';

;WITH L AS (
    SELECT sp.name, sp.type_desc, sp.is_disabled, sp.default_database_name, sp.create_date
    FROM sys.server_principals sp
    WHERE sp.name = @Login
)
SELECT
    @@SERVERNAME                          AS ServerName,
    CAST(SERVERPROPERTY('MachineName') AS nvarchar(128)) AS MachineName,
    CAST(SERVERPROPERTY('InstanceName') AS nvarchar(128)) AS InstanceName,
    CAST(SERVERPROPERTY('ProductVersion') AS nvarchar(50)) AS ProductVersion,
    CAST(SERVERPROPERTY('Edition') AS nvarchar(200)) AS Edition,
    @Login                                AS LoginName,
    CASE WHEN EXISTS (SELECT 1 FROM L) THEN 1 ELSE 0 END AS LoginExists,
    (SELECT TOP(1) type_desc FROM L)       AS LoginType,
    (SELECT TOP(1) is_disabled FROM L)     AS IsDisabled,
    ISNULL(IS_SRVROLEMEMBER(N'sysadmin', @Login), 0) AS IsSysadmin,
    STUFF((
        SELECT N', ' + r.name
        FROM sys.server_role_members rm
        JOIN sys.server_principals r ON r.principal_id = rm.role_principal_id
        JOIN sys.server_principals m ON m.principal_id = rm.member_principal_id
        WHERE m.name = @Login
        ORDER BY r.name
        FOR XML PATH(''), TYPE).value('.', 'nvarchar(max)')
    ,1,2,N'') AS ServerRoles;
"@

$results = @()

foreach ($s in $servers) {
  try {
    $params = @{
      ServerInstance = $s
      Database       = "master"
      Query          = $tsql
      Variable       = @("LoginName=$LoginName")
      TrustServerCertificate = $true
      ErrorAction    = "Stop"
    }

    if (-not $UseIntegratedSecurity) {
      if (-not $SqlUser -or -not $SqlPassword) { throw "SQL auth: podaj -SqlUser i -SqlPassword albo użyj Windows auth." }
      $sec = ConvertTo-SecureString $SqlPassword -AsPlainText -Force
      $cred = New-Object System.Management.Automation.PSCredential($SqlUser, $sec)
      $params.Credential = $cred
    }

    $row = Invoke-Sqlcmd @params | Select-Object -First 1
    if (-not $row) {
      $results += [pscustomobject]@{
        Server = $s; Status="NO_DATA"; Error="Brak danych (zapytanie nie zwróciło wiersza)"; LoginExists=$null; IsSysadmin=$null
      }
    } else {
      $results += [pscustomobject]@{
        Server        = $s
        Status        = "OK"
        Error         = ""
        ServerName    = $row.ServerName
        MachineName   = $row.MachineName
        InstanceName  = $row.InstanceName
        ProductVersion= $row.ProductVersion
        Edition       = $row.Edition
        LoginName     = $row.LoginName
        LoginExists   = [int]$row.LoginExists
        LoginType     = $row.LoginType
        IsDisabled    = $row.IsDisabled
        IsSysadmin    = [int]$row.IsSysadmin
        ServerRoles   = $row.ServerRoles
      }
    }
  }
  catch {
    $results += [pscustomobject]@{
      Server = $s; Status="FAIL"; Error=$_.Exception.Message; LoginExists=$null; IsSysadmin=$null
    }
  }
}

# --- eksport CSV ---
$ts = Get-Date -Format "yyyyMMdd_HHmmss"
$csvPath  = Join-Path $OutDir "audit_sysadmin_$($LoginName -replace '[\\/:*?""<>|]','_')_$ts.csv"
$results | Export-Csv -NoTypeInformation -Encoding UTF8 -Path $csvPath

# --- HTML (tabela + sort/filter via DataTables) ---
$htmlPath = Join-Path $OutDir "audit_sysadmin_$($LoginName -replace '[\\/:*?""<>|]','_')_$ts.html"

$rowsJson = ($results | ConvertTo-Json -Depth 5)

$html = @"
<!doctype html>
<html lang="pl">
<head>
<meta charset="utf-8"/>
<title>Audit sysadmin — $LoginName</title>
<link rel="stylesheet" href="https://cdn.datatables.net/1.13.8/css/jquery.dataTables.min.css">
<script src="https://code.jquery.com/jquery-3.7.1.min.js"></script>
<script src="https://cdn.datatables.net/1.13.8/js/jquery.dataTables.min.js"></script>
<style>
  body { font-family: Segoe UI, Arial, sans-serif; margin: 16px; }
  .pill { padding: 2px 8px; border-radius: 999px; display:inline-block; }
  .ok { background:#e7f7ee; }
  .fail { background:#fde7e7; }
  .warn { background:#fff4db; }
  table.dataTable tbody td { vertical-align: top; }
  .small { color:#666; font-size: 12px; }
</style>
</head>
<body>
<h2>Audit sysadmin — <code>$LoginName</code></h2>
<div class="small">Wygenerowano: $(Get-Date)</div>
<hr/>

<table id="t" class="display" style="width:100%">
  <thead>
    <tr>
      <th>Status</th>
      <th>Server</th>
      <th>ServerName</th>
      <th>LoginExists</th>
      <th>IsSysadmin</th>
      <th>IsDisabled</th>
      <th>LoginType</th>
      <th>ServerRoles</th>
      <th>Version</th>
      <th>Edition</th>
      <th>Error</th>
    </tr>
  </thead>
  <tbody></tbody>
</table>

<script>
const data = $rowsJson;

function pill(status){
  if(status === "OK") return '<span class="pill ok">OK</span>';
  if(status === "FAIL") return '<span class="pill fail">FAIL</span>';
  return '<span class="pill warn">'+status+'</span>';
}

\$(document).ready(function(){
  \$('#t').DataTable({
    data: data,
    pageLength: 50,
    order: [[4,'desc'], [3,'desc'], [1,'asc']],
    columns: [
      { data: 'Status', render: (d)=>pill(d) },
      { data: 'Server' },
      { data: 'ServerName' },
      { data: 'LoginExists' },
      { data: 'IsSysadmin' },
      { data: 'IsDisabled' },
      { data: 'LoginType' },
      { data: 'ServerRoles' },
      { data: 'ProductVersion' },
      { data: 'Edition' },
      { data: 'Error' }
    ]
  });
});
</script>

</body>
</html>
"@

Set-Content -Path $htmlPath -Value $html -Encoding UTF8

Write-Host "DONE"
Write-Host "CSV : $csvPath"
Write-Host "HTML: $htmlPath"

# --- szybkie podsumowanie w konsoli ---
$ok = $results | Where-Object Status -eq "OK"
Write-Host "`nSummary:"
Write-Host ("  OK: {0}, FAIL: {1}" -f ($ok.Count), (($results | Where-Object Status -ne "OK").Count))
Write-Host ("  Login exists on: {0}" -f (($ok | Where-Object LoginExists -eq 1).Count))
Write-Host ("  Sysadmin on:     {0}" -f (($ok | Where-Object IsSysadmin -eq 1).Count))
